class S:
    """One scene. Drawing happens in window coordinates (0..SW, 0..SH)."""

    def __init__(self, seed):
        self.r = random.Random(seed)
        self.defs = []
        self.body = []
        self.n = 0

    def id(self, p="d"):
        self.n += 1
        return f"{p}{self.n}"

    def add(self, s):
        self.body.append(s)

    # ---------- paint ----------
    def lin(self, stops, x1=0, y1=0, x2=0, y2=1, units="objectBoundingBox"):
        gid = self.id("lg")
        st = "".join(f'<stop offset="{o}" stop-color="{c}" stop-opacity="{a}"/>' for o, c, a in stops)
        self.defs.append(f'<linearGradient id="{gid}" x1="{x1}" y1="{y1}" x2="{x2}" y2="{y2}" '
                         f'gradientUnits="{units}">{st}</linearGradient>')
        return f"url(#{gid})"

    def rad(self, stops, cx, cy, r):
        gid = self.id("rg")
        st = "".join(f'<stop offset="{o}" stop-color="{c}" stop-opacity="{a}"/>' for o, c, a in stops)
        self.defs.append(f'<radialGradient id="{gid}" cx="{cx}" cy="{cy}" r="{r}" '
                         f'gradientUnits="userSpaceOnUse">{st}</radialGradient>')
        return f"url(#{gid})"

    def blur(self, sd):
        fid = self.id("bl")
        self.defs.append(f'<filter id="{fid}" x="-50%" y="-50%" width="200%" height="200%">'
                         f'<feGaussianBlur stdDeviation="{sd}"/></filter>')
        return f"url(#{fid})"

    # ---------- building blocks ----------
    def sky(self, *colors):
        n = len(colors) - 1
        stops = [(i / n, c, 1) for i, c in enumerate(colors)]
        self.add(f'<rect width="{SW}" height="{SH}" fill="{self.lin(stops)}"/>')

    def glow(self, x, y, r, color="#ffffff", a=0.5):
        f = self.rad([(0, color, a), (0.45, color, a * 0.4), (1, color, 0)], x, y, r)
        self.add(f'<circle cx="{x}" cy="{y}" r="{r}" fill="{f}"/>')

    def sun(self, x, y, r, c1="#fff4c2", c2="#ffd166", halo="#ffcf70", ha=0.55):
        self.glow(x, y, r * 3.2, halo, ha)
        f = self.rad([(0, c1, 1), (1, c2, 1)], x, y, r)
        self.add(f'<circle cx="{x}" cy="{y}" r="{r}" fill="{f}"/>')

    def moon(self, x, y, r, crescent=False, color="#f6eecf", halo="#fff3c4", ha=0.35):
        self.glow(x, y, r * 3, halo, ha)
        if crescent:
            mid = self.id("mk")
            self.defs.append(f'<mask id="{mid}"><rect x="{x - r - 2}" y="{y - r - 2}" width="{2 * r + 4}" '
                             f'height="{2 * r + 4}" fill="#fff"/><circle cx="{x + r * 0.45}" cy="{y - r * 0.25}" '
                             f'r="{r * 0.92}" fill="#000"/></mask>')
            self.add(f'<circle cx="{x}" cy="{y}" r="{r}" fill="{color}" mask="url(#{mid})"/>')
        else:
            self.add(f'<circle cx="{x}" cy="{y}" r="{r}" fill="{color}"/>')
            for dx, dy, rr in ((-0.3, -0.2, 0.2), (0.25, 0.25, 0.15), (-0.05, 0.45, 0.09), (0.35, -0.35, 0.08)):
                self.add(f'<circle cx="{x + dx * r}" cy="{y + dy * r}" r="{rr * r}" fill="#000" opacity="0.06"/>')

    def stars(self, n, y_max, color="#ffffff", big=0.12):
        for _ in range(n):
            x, y = self.r.uniform(10, SW - 10), self.r.uniform(8, y_max)
            if self.r.random() < big:
                self.sparkle(x, y, self.r.uniform(6, 10), color, self.r.uniform(0.7, 1))
            else:
                self.add(f'<circle cx="{x:.1f}" cy="{y:.1f}" r="{self.r.uniform(0.9, 2.2):.1f}" fill="{color}" '
                         f'opacity="{self.r.uniform(0.45, 1):.2f}"/>')

    def sparkle(self, x, y, r, color="#ffffff", a=1.0):
        k = r * 0.22
        self.add(f'<path d="M{x},{y - r} Q{x + k},{y - k} {x + r},{y} Q{x + k},{y + k} {x},{y + r} '
                 f'Q{x - k},{y + k} {x - r},{y} Q{x - k},{y - k} {x},{y - r}Z" fill="{color}" opacity="{a:.2f}"/>')

    def ridge(self, y0, amp, waves, phase, fill, jag=0.0, pts=140):
        d = [f"M0,{SH}"]
        for i in range(pts + 1):
            t = i / pts
            y = (y0 + amp * math.sin(2 * math.pi * waves * t + phase)
                 + amp * 0.35 * math.sin(2 * math.pi * waves * 2.7 * t + phase * 1.9))
            if jag:
                y += self.r.uniform(-jag, jag)
            d.append(f"L{SW * t:.1f},{y:.1f}")
        d.append(f"L{SW},{SH}Z")
        self.add(f'<path d="{" ".join(d)}" fill="{fill}"/>')

    def ridge_y(self, y0, amp, waves, phase, x):
        t = x / SW
        return (y0 + amp * math.sin(2 * math.pi * waves * t + phase)
                + amp * 0.35 * math.sin(2 * math.pi * waves * 2.7 * t + phase * 1.9))

    def peaks(self, base, pts, fill, snow=None):
        d = f"M0,{SH} L0,{base} " + " ".join(f"L{x},{y}" for x, y in pts) + f" L{SW},{base} L{SW},{SH}Z"
        self.add(f'<path d="{d}" fill="{fill}"/>')
        if snow:
            for (x, y) in pts:
                if y < base - 120:
                    w = (base - y) * 0.28
                    self.add(f'<path d="M{x},{y} L{x - w},{y + w * 1.1} L{x - w * 0.45},{y + w * 0.85} '
                             f'L{x - w * 0.1},{y + w * 1.2} L{x + w * 0.35},{y + w * 0.9} L{x + w},{y + w * 1.15}Z" '
                             f'fill="{snow}"/>')

    def pine(self, x, base, h, color, snow=None):
        w = h * 0.46
        self.add(f'<rect x="{x - h * 0.03}" y="{base - h * 0.12}" width="{h * 0.06}" height="{h * 0.14}" fill="{color}"/>')
        for k, (top, wid) in enumerate(((0.0, 0.55), (0.25, 0.8), (0.5, 1.0))):
            ty = base - h + h * top
            by = ty + h * 0.48
            hw = w * wid / 2
            self.add(f'<path d="M{x},{ty} L{x + hw},{by} L{x - hw},{by}Z" fill="{color}"/>')
            if snow:
                self.add(f'<path d="M{x},{ty} L{x + hw * 0.38},{ty + h * 0.18} Q{x},{ty + h * 0.13} '
                         f'{x - hw * 0.38},{ty + h * 0.18}Z" fill="{snow}"/>')

    def round_tree(self, x, base, h, color, trunk="#3b2a1e"):
        self.add(f'<rect x="{x - h * 0.04}" y="{base - h * 0.42}" width="{h * 0.08}" height="{h * 0.42}" '
                 f'rx="{h * 0.03}" fill="{trunk}"/>')
        self.add(f'<ellipse cx="{x}" cy="{base - h * 0.62}" rx="{h * 0.3}" ry="{h * 0.38}" fill="{color}"/>')

    def bare_tree(self, x, base, h, color, depth=5, width=None):
        width = width or h * 0.07

        def branch(x0, y0, ang, length, w, d):
            x1 = x0 + length * math.cos(ang)
            y1 = y0 - length * math.sin(ang)
            self.add(f'<line x1="{x0:.1f}" y1="{y0:.1f}" x2="{x1:.1f}" y2="{y1:.1f}" stroke="{color}" '
                     f'stroke-width="{max(w, 1.2):.1f}" stroke-linecap="round"/>')
            if d <= 0:
                return
            for da in (-0.5, 0.42):
                branch(x1, y1, ang + da + self.r.uniform(-0.18, 0.18), length * self.r.uniform(0.62, 0.75),
                       w * 0.62, d - 1)

        branch(x, base, math.pi / 2 + self.r.uniform(-0.08, 0.08), h * 0.36, width, depth)

    def bat(self, x, y, s, color="#140f1f", rot=0):
        self.add(f'<g transform="translate({x},{y}) rotate({rot}) scale({s})"><path d="M0,-2 C-5,-8 -13,-9 -22,-5 '
                 f'C-19,-3 -18,1 -20,5 C-15,3 -11,4 -8,7 C-6,4 -3,3 -1.5,4.5 L0,3 L1.5,4.5 C3,3 6,4 8,7 C11,4 15,3 '
                 f'20,5 C18,1 19,-3 22,-5 C13,-9 5,-8 0,-2Z M-2,-3 L-2.6,-6.5 L-0.8,-4 L0.8,-4 L2.6,-6.5 L2,-3Z" '
                 f'fill="{color}"/></g>')

    def bird(self, x, y, s, color="#2b2b2b", a=0.8):
        self.add(f'<path d="M{x - 10 * s},{y} Q{x - 5 * s},{y - 6 * s} {x},{y} Q{x + 5 * s},{y - 6 * s} {x + 10 * s},{y}" '
                 f'fill="none" stroke="{color}" stroke-width="{2.2 * s}" stroke-linecap="round" opacity="{a}"/>')

    def heart_path(self, x, y, s):
        p = [(0, 0.32), (-0.52, -0.05), (-0.52, -0.32), (-0.52, -0.6), (-0.24, -0.68), (0, -0.44),
             (0.24, -0.68), (0.52, -0.6), (0.52, -0.32), (0.52, -0.05), (0, 0.32)]
        q = [(x + px * s, y + py * s) for px, py in p]
        return (f"M{q[0][0]:.1f},{q[0][1]:.1f} C{q[1][0]:.1f},{q[1][1]:.1f} {q[2][0]:.1f},{q[2][1]:.1f} "
                f"{q[2][0]:.1f},{q[2][1]:.1f} C{q[3][0]:.1f},{q[3][1]:.1f} {q[4][0]:.1f},{q[4][1]:.1f} "
                f"{q[5][0]:.1f},{q[5][1]:.1f} C{q[6][0]:.1f},{q[6][1]:.1f} {q[7][0]:.1f},{q[7][1]:.1f} "
                f"{q[8][0]:.1f},{q[8][1]:.1f} C{q[9][0]:.1f},{q[9][1]:.1f} {q[10][0]:.1f},{q[10][1]:.1f} "
                f"{q[10][0]:.1f},{q[10][1]:.1f}Z")

    def heart(self, x, y, s, color, a=1.0, rot=0):
        self.add(f'<g transform="rotate({rot} {x} {y})"><path d="{self.heart_path(x, y, s)}" fill="{color}" '
                 f'opacity="{a}"/></g>')

    def balloon_heart(self, x, y, s, c1, c2, string_to):
        f = self.lin([(0, c1, 1), (1, c2, 1)])
        sx, sy = string_to
        self.add(f'<path d="M{x},{y + 0.3 * s} C{x - 0.15 * s},{(y + sy) / 2} {sx + 0.12 * s},{(y + sy) / 2} {sx},{sy}" '
                 f'fill="none" stroke="#ffffff" stroke-opacity="0.75" stroke-width="2"/>')
        self.add(f'<path d="{self.heart_path(x, y, s)}" fill="{f}"/>')
        self.add(f'<path d="M{x - 0.32 * s},{y - 0.42 * s} Q{x - 0.4 * s},{y - 0.2 * s} {x - 0.28 * s},{y - 0.08 * s}" '
                 f'fill="none" stroke="#ffffff" stroke-opacity="0.55" stroke-width="{0.05 * s}" stroke-linecap="round"/>')

    def cloud(self, x, y, s, color="#ffffff", a=0.95):
        self.add(f'<g opacity="{a}" fill="{color}"><circle cx="{x - 0.5 * s}" cy="{y}" r="{0.35 * s}"/>'
                 f'<circle cx="{x}" cy="{y - 0.2 * s}" r="{0.5 * s}"/><circle cx="{x + 0.55 * s}" cy="{y + 0.02 * s}" '
                 f'r="{0.33 * s}"/><rect x="{x - 0.85 * s}" y="{y}" width="{1.7 * s}" height="{0.35 * s}" '
                 f'rx="{0.17 * s}"/></g>')

    def burst(self, x, y, r, c1, c2, rays=22, a=1.0):
        self.glow(x, y, r * 1.4, c1, 0.35 * a)
        for i in range(rays):
            ang = 2 * math.pi * i / rays + self.r.uniform(-0.04, 0.04)
            r1, r2 = r * self.r.uniform(0.18, 0.3), r * self.r.uniform(0.82, 1.0)
            x1, y1 = x + r1 * math.cos(ang), y + r1 * math.sin(ang)
            x2, y2 = x + r2 * math.cos(ang), y + r2 * math.sin(ang)
            f = self.lin([(0, c2, 0.0), (1, c1, a)], x1, y1, x2, y2, "userSpaceOnUse")
            self.add(f'<line x1="{x1:.1f}" y1="{y1:.1f}" x2="{x2:.1f}" y2="{y2:.1f}" stroke="{f}" '
                     f'stroke-width="{max(2.0, r * 0.03):.1f}" stroke-linecap="round"/>')
            self.add(f'<circle cx="{x2 + r * 0.07 * math.cos(ang):.1f}" cy="{y2 + r * 0.07 * math.sin(ang):.1f}" '
                     f'r="{max(1.8, r * 0.022):.1f}" fill="{c2}" opacity="{a}"/>')

    def snow(self, n, y0=0, y1=SH, a=(0.5, 0.95), size=(1.5, 4.5)):
        for _ in range(n):
            self.add(f'<circle cx="{self.r.uniform(0, SW):.1f}" cy="{self.r.uniform(y0, y1):.1f}" '
                     f'r="{self.r.uniform(*size):.1f}" fill="#ffffff" opacity="{self.r.uniform(*a):.2f}"/>')

    def leaf(self, x, y, s, color, rot):
        self.add(f'<g transform="translate({x},{y}) rotate({rot}) scale({s})"><path d="M0,-10 C6,-6 7,4 0,10 '
                 f'C-7,4 -6,-6 0,-10Z" fill="{color}"/><path d="M0,-8 L0,12" stroke="#00000033" stroke-width="1"/></g>')

    def petal(self, x, y, s, color, rot, a=0.95):
        self.add(f'<ellipse cx="{x}" cy="{y}" rx="{6 * s}" ry="{3.6 * s}" fill="{color}" opacity="{a}" '
                 f'transform="rotate({rot} {x} {y})"/>')

    def blossom(self, x, y, r, color, center="#ffe08a"):
        for i in range(5):
            a = 2 * math.pi * i / 5 - math.pi / 2
            self.add(f'<circle cx="{x + r * 0.55 * math.cos(a):.1f}" cy="{y + r * 0.55 * math.sin(a):.1f}" '
                     f'r="{r * 0.5:.1f}" fill="{color}"/>')
        self.add(f'<circle cx="{x}" cy="{y}" r="{r * 0.28:.1f}" fill="{center}"/>')

    def tulip(self, x, base, h, color, stem="#4f8f4a"):
        self.add(f'<path d="M{x},{base} Q{x + 4},{base - h * 0.5} {x},{base - h * 0.78}" fill="none" stroke="{stem}" '
                 f'stroke-width="{h * 0.05:.1f}" stroke-linecap="round"/>')
        self.add(f'<path d="M{x},{base - h * 0.2} Q{x + h * 0.32},{base - h * 0.4} {x + h * 0.26},{base - h * 0.62} '
                 f'Q{x + h * 0.08},{base - h * 0.45} {x},{base - h * 0.2}Z" fill="{stem}"/>')
        cy = base - h * 0.86
        w = h * 0.2
        self.add(f'<path d="M{x - w},{cy - w * 0.9} L{x - w * 0.45},{cy - w * 0.35} L{x},{cy - w * 1.05} '
                 f'L{x + w * 0.45},{cy - w * 0.35} L{x + w},{cy - w * 0.9} Q{x + w * 1.05},{cy + w * 0.9} {x},{cy + w * 0.95} '
                 f'Q{x - w * 1.05},{cy + w * 0.9} {x - w},{cy - w * 0.9}Z" fill="{color}"/>')

    def egg(self, x, y, s, c1, c2, rot=0):
        cid = self.id("eg")
        d = f"M{x},{y - s} C{x + s * 0.72},{y - s} {x + s * 0.8},{y + s * 0.85} {x},{y + s * 0.85} " \
            f"C{x - s * 0.8},{y + s * 0.85} {x - s * 0.72},{y - s} {x},{y - s}Z"
        self.defs.append(f'<clipPath id="{cid}"><path d="{d}"/></clipPath>')
        zig = " ".join(f"L{x - s + i * s * 0.25:.1f},{y + (s * 0.12 if i % 2 else -s * 0.12):.1f}" for i in range(9))
        self.add(f'<g transform="rotate({rot} {x} {y})"><path d="{d}" fill="{c1}"/><g clip-path="url(#{cid})">'
                 f'<path d="M{x - s},{y} {zig}" fill="none" stroke="{c2}" stroke-width="{s * 0.16:.1f}" '
                 f'stroke-linejoin="round"/><circle cx="{x - s * 0.3}" cy="{y - s * 0.45}" r="{s * 0.1}" fill="{c2}"/>'
                 f'<circle cx="{x + s * 0.25}" cy="{y - s * 0.55}" r="{s * 0.08}" fill="{c2}"/>'
                 f'<circle cx="{x}" cy="{y + s * 0.45}" r="{s * 0.1}" fill="{c2}"/></g>'
                 f'<path d="{d}" fill="none" stroke="#ffffff" stroke-opacity="0.25" stroke-width="2"/></g>')

    def candle(self, x, base, h, w, color="#f4ead7"):
        self.glow(x, base - h - w * 0.9, w * 3.4, "#ffb347", 0.55)
        self.add(f'<rect x="{x - w / 2}" y="{base - h}" width="{w}" height="{h}" rx="{w * 0.18}" fill="{color}"/>')
        self.add(f'<path d="M{x},{base - h - w * 1.5} Q{x + w * 0.42},{base - h - w * 0.55} {x},{base - h - w * 0.1} '
                 f'Q{x - w * 0.42},{base - h - w * 0.55} {x},{base - h - w * 1.5}Z" fill="#ffcf5a"/>')

    def papel(self, x0, x1, y, colors, h=92):
        n = len(colors)
        sag = 26
        self.add(f'<path d="M{x0 - 20},{y - 6} Q{(x0 + x1) / 2},{y + sag} {x1 + 20},{y - 6}" fill="none" '
                 f'stroke="#f5f0e6" stroke-width="3"/>')
        w = (x1 - x0) / n * 0.86
        for i, c in enumerate(colors):
            cx = x0 + (x1 - x0) * (i + 0.5) / n
            t = (cx - x0) / (x1 - x0)
            top = y - 6 + 4 * (sag + 6) * t * (1 - t) * 0.5 + 2
            l, r_ = cx - w / 2, cx + w / 2
            b = top + h
            zig = " ".join(f"L{r_ - k * w / 6:.1f},{b + (8 if k % 2 else 0):.1f}" for k in range(7))
            outer = f"M{l},{top} L{r_},{top} L{r_},{b} {zig} L{l},{top}Z"
            holes = (f" M{cx},{top + h * 0.22} L{cx + w * 0.18},{top + h * 0.45} L{cx},{top + h * 0.68} "
                     f"L{cx - w * 0.18},{top + h * 0.45}Z M{l + w * 0.12},{top + h * 0.14} L{l + w * 0.24},{top + h * 0.14} "
                     f"L{l + w * 0.24},{top + h * 0.26} L{l + w * 0.12},{top + h * 0.26}Z M{r_ - w * 0.24},{top + h * 0.14} "
                     f"L{r_ - w * 0.12},{top + h * 0.14} L{r_ - w * 0.12},{top + h * 0.26} L{r_ - w * 0.24},{top + h * 0.26}Z")
            self.add(f'<path d="{outer}{holes}" fill="{c}" fill-rule="evenodd" '
                     f'transform="rotate({self.r.uniform(-3, 3):.1f} {cx} {top})"/>')

    def bunting(self, x0, x1, y, colors, size=46, sag=40):
        self.add(f'<path d="M{x0},{y} Q{(x0 + x1) / 2},{y + sag * 2} {x1},{y}" fill="none" stroke="#f7f1e3" '
                 f'stroke-width="3"/>')
        n = len(colors)
        for i, c in enumerate(colors):
            t = (i + 0.5) / n
            cx = x0 + (x1 - x0) * t
            cy = y + 2 * sag * 2 * t * (1 - t) * 0.5 * 2 * 0.5 + 2
            self.add(f'<path d="M{cx - size / 2},{cy} L{cx + size / 2},{cy} L{cx},{cy + size * 1.1}Z" fill="{c}"/>')

    def skyline(self, base, color, lit="#ffd27a", seed_density=0.35, heights=(90, 260)):
        x = -10
        while x < SW + 10:
            w = self.r.uniform(48, 96)
            h = self.r.uniform(*heights)
            self.add(f'<rect x="{x:.1f}" y="{base - h:.1f}" width="{w:.1f}" height="{h + 2:.1f}" fill="{color}"/>')
            if self.r.random() < 0.25:
                self.add(f'<rect x="{x + w * 0.42:.1f}" y="{base - h - 26:.1f}" width="3" height="26" fill="{color}"/>')
            for wy in range(int(base - h + 14), int(base - 12), 18):
                for wx in range(int(x + 9), int(x + w - 12), 15):
                    if self.r.random() < seed_density:
                        self.add(f'<rect x="{wx}" y="{wy}" width="6" height="8" fill="{lit}" '
                                 f'opacity="{self.r.uniform(0.55, 1):.2f}"/>')
            x += w + self.r.uniform(2, 10)

    def water(self, y, c1, c2, shimmer="#ffffff"):
        f = self.lin([(0, c1, 1), (1, c2, 1)])
        self.add(f'<rect x="0" y="{y}" width="{SW}" height="{SH - y}" fill="{f}"/>')
        for _ in range(26):
            yy = self.r.uniform(y + 8, SH - 6)
            ww = self.r.uniform(30, 120)
            xx = self.r.uniform(0, SW)
            self.add(f'<rect x="{xx:.1f}" y="{yy:.1f}" width="{ww:.1f}" height="2.5" rx="1.2" fill="{shimmer}" '
                     f'opacity="{self.r.uniform(0.12, 0.35):.2f}"/>')

    def aurora(self, colors, y=170, a=0.75):
        f = self.blur(22)
        for i, c in enumerate(colors):
            pts = []
            for k in range(0, 13):
                x = SW * k / 12
                pts.append((x, y + i * 26 + 46 * math.sin(k * 0.9 + i * 1.3)))
            d = "M" + " L".join(f"{x:.1f},{yy:.1f}" for x, yy in pts)
            d2 = " L".join(f"{x:.1f},{yy + 130:.1f}" for x, yy in reversed(pts))
            grad = self.lin([(0, c, 0.0), (0.35, c, a), (1, c, 0.0)])
            self.add(f'<path d="{d} L{d2}Z" fill="{grad}" filter="{f}"/>')

    def mist(self, y, a=0.25):
        f = self.blur(18)
        self.add(f'<ellipse cx="{SW * 0.3}" cy="{y}" rx="{SW * 0.45}" ry="26" fill="#ffffff" opacity="{a}" filter="{f}"/>')
        self.add(f'<ellipse cx="{SW * 0.75}" cy="{y + 18}" rx="{SW * 0.4}" ry="22" fill="#ffffff" opacity="{a * 0.8}" filter="{f}"/>')
